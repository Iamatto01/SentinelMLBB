function checkOcrSetup() {
  const ui = SpreadsheetApp.getUi();
  const lines = [];

  // Check Advanced Drive Service
  const advDrive = (typeof Drive !== 'undefined' && Drive.Files && typeof Drive.Files.insert === 'function');
  lines.push('Advanced Drive service: ' + (advDrive ? 'ENABLED' : 'DISABLED'));

  // Check DriveApp access
  try {
    DriveApp.getRootFolder().getId();
    lines.push('DriveApp access: OK');
  } catch (e) {
    lines.push('DriveApp access: ERROR - ' + e.message);
  }

  // Check Drive REST API reachable with current token
  try {
    const token = ScriptApp.getOAuthToken();
    const resp = UrlFetchApp.fetch('https://www.googleapis.com/drive/v3/about?fields=user', {
      method: 'get',
      muteHttpExceptions: true,
      headers: { Authorization: 'Bearer ' + token }
    });
    const code = resp.getResponseCode();
    if (code === 200) lines.push('Drive REST API: reachable (200 OK)');
    else lines.push('Drive REST API: HTTP ' + code + ' — ' + (resp.getContentText() || '').slice(0, 300));
  } catch (e) {
    lines.push('Drive REST API: request failed - ' + e.message);
  }

  // Guidance
  lines.push('');
  if (!advDrive || lines.some(l => /ERROR|DISABLED|failed|HTTP\s4/i.test(l))) {
    lines.push('ACTION: To fix OCR:');
    lines.push('1) In Apps Script editor -> Services -> Add "Drive API" (Advanced Google services).');
    lines.push('2) In Google Cloud Console for this project -> APIs & Services -> Library -> Enable "Google Drive API".');
    lines.push('3) If you modified scopes, open Project Settings -> Show "appsscript.json" and include Drive scopes, then re-authorize.');
    lines.push('4) After enabling, run "📷 OCR Upload" once and allow all permissions.');
  } else {
    lines.push('OK: OCR setup looks ready. Try uploading a screenshot via MLBB System > 📷 OCR Upload to verify.');
  }

  ui.alert(lines.join('\n'));
}

function uploadOcrScreenshot(payload) {
  ensureOcrReady_();

  const p = payload || {};
  const fileName = String(p.fileName || "upload.png");
  const mimeType = String(p.mimeType || "image/png");
  const dataBase64 = String(p.dataBase64 || "").trim();
  if (!dataBase64) throw new Error("Empty file payload.");

  const bytes = Utilities.base64Decode(dataBase64);
  const blob = Utilities.newBlob(bytes, mimeType, fileName);

  const ss = getSafeActiveSpreadsheet();
  const inbox = ss.getSheetByName("📥 OCR Inbox");
  if (!inbox) throw new Error("OCR Inbox not found. Run Setup OCR Inbox first.");

  const rowNumber = findNextOcrInboxRow_();
  const savedFile = DriveApp.createFile(blob).setName("OCR_UPLOAD_" + Date.now() + "_" + fileName);

  inbox.getRange(rowNumber, 1).setValue(new Date());
  inbox.getRange(rowNumber, 2).setValue(savedFile.getUrl());
  inbox.getRange(rowNumber, 3).setValue("Pending");
  inbox.getRange(rowNumber, 4, 1, 4).clearContent();

  try {
    const rawText = extractTextFromImageBlob_(savedFile.getBlob());
    inbox.getRange(rowNumber, 5).setValue(rawText);

    const parsed = parsePostgameText_(rawText);
    const summary = "Mode=" + (parsed.mode || "?") + ", Result=" + (parsed.result || "?") + ", Duration=" + (parsed.duration || 0) + "m, Pairs=" + parsed.pairs.length;
    inbox.getRange(rowNumber, 4).setValue(summary);

    if (!parsed.result || parsed.pairs.length === 0) {
      inbox.getRange(rowNumber, 3).setValue("Review");
      return "Uploaded. OCR saved to Inbox row " + rowNumber + " (Review needed).";
    }

    const insertedRow = appendParsedGameToLog_(parsed);
    inbox.getRange(rowNumber, 3).setValue("Done");
    inbox.getRange(rowNumber, 7).setValue(insertedRow);
    return "Uploaded and processed. Inbox row " + rowNumber + ", Game Log row " + insertedRow + ".";
  } catch (err) {
    inbox.getRange(rowNumber, 3).setValue("Error");
    inbox.getRange(rowNumber, 6).setValue(String(err && err.message ? err.message : err));
    return "Upload saved, but OCR failed at Inbox row " + rowNumber + ".";
  }
}

function findNextOcrInboxRow_() {
  const sh = getSafeActiveSpreadsheet().getSheetByName("📥 OCR Inbox");
  const values = sh.getRange(4, 2, 200, 1).getValues();
  for (let i = 0; i < values.length; i++) {
    if (!values[i][0]) return i + 4;
  }
  throw new Error("OCR Inbox is full (200 rows).");
}

function processPendingOcrInbox() {
  ensureOcrReady_();

  const ss = getSafeActiveSpreadsheet();
  const sh = ss.getSheetByName("📥 OCR Inbox");
  if (!sh) {
    SpreadsheetApp.getUi().alert("OCR Inbox not found. Run Setup OCR Inbox first.");
    return;
  }

  const rows = sh.getRange(4, 1, 200, 7).getValues();
  let done = 0, review = 0, failed = 0;
  rows.forEach((row, idx) => {
    const status = String(row[2] || "").trim();
    const link = String(row[1] || "").trim();
    if (!link) return;
    if (status && status !== "Pending") return;
    const out = processSingleOcrInboxRow_(idx + 4, link);
    if (out === "Done") done++;
    else if (out === "Review") review++;
    else failed++;
  });

  SpreadsheetApp.getUi().alert("OCR done. Success: " + done + " | Review: " + review + " | Error: " + failed);
}

function getImageBlobFromInput_(imageInput) {
  const input = String(imageInput || "").trim();
  if (!input) throw new Error("Image link / file ID is empty.");

  const driveId = extractDriveFileId_(input);
  if (driveId) {
    return DriveApp.getFileById(driveId).getBlob();
  }

  if (/^https?:\/\//i.test(input)) {
    const resp = UrlFetchApp.fetch(input, { muteHttpExceptions: true });
    if (resp.getResponseCode() >= 400) {
      throw new Error("Unable to download image URL. HTTP " + resp.getResponseCode());
    }
    return resp.getBlob();
  }

  return DriveApp.getFileById(input).getBlob();
}

function extractDriveFileId_(text) {
  const s = String(text || "");
  const p1 = s.match(/\/d\/([a-zA-Z0-9_-]{20,})/);
  if (p1) return p1[1];
  const p2 = s.match(/[?&]id=([a-zA-Z0-9_-]{20,})/);
  if (p2) return p2[1];
  if (/^[a-zA-Z0-9_-]{20,}$/.test(s)) return s;
  return "";
}

function extractTextFromImageBlob_(blob) {
  if (typeof Drive !== "undefined" && Drive.Files && typeof Drive.Files.insert === "function") {
    try {
      const resource = { title: "OCR_TMP_" + Date.now(), mimeType: MimeType.GOOGLE_DOCS };
      const doc = Drive.Files.insert(resource, blob, { ocr: true, ocrLanguage: "en" });
      const text = DocumentApp.openById(doc.id).getBody().getText() || "";
      DriveApp.getFileById(doc.id).setTrashed(true);
      return text;
    } catch (e) {
      throw new Error("OCR gagal diproses. Pastikan Drive API aktif dan file image boleh diakses. Detail: " + e.message);
    }
  }

  return extractTextViaDriveRest_(blob);
}

function extractTextViaDriveRest_(blob) {
  const token = ScriptApp.getOAuthToken();
  const boundary = "mlbb_ocr_boundary_" + Date.now();
  const metadata = {
    title: "OCR_TMP_" + Date.now(),
    mimeType: "application/vnd.google-apps.document"
  };

  const delimiter = "\r\n--" + boundary + "\r\n";
  const closeDelim = "\r\n--" + boundary + "--";
  const body =
    delimiter +
    "Content-Type: application/json; charset=UTF-8\r\n\r\n" +
    JSON.stringify(metadata) +
    delimiter +
    "Content-Type: " + (blob.getContentType() || "image/png") + "\r\n" +
    "Content-Transfer-Encoding: base64\r\n\r\n" +
    Utilities.base64Encode(blob.getBytes()) +
    closeDelim;

  const url = "https://www.googleapis.com/upload/drive/v2/files?uploadType=multipart&ocr=true&ocrLanguage=en&convert=true";
  const resp = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "multipart/related; boundary=" + boundary,
    headers: { Authorization: "Bearer " + token },
    payload: body,
    muteHttpExceptions: true
  });

  if (resp.getResponseCode() >= 400) {
    throw new Error("Drive OCR REST gagal (HTTP " + resp.getResponseCode() + "). Aktifkan Google Drive API pada Apps Script project.");
  }

  const json = JSON.parse(resp.getContentText() || "{}");
  const docId = json.id;
  if (!docId) {
    throw new Error("OCR response tiada document id.");
  }

  const text = DocumentApp.openById(docId).getBody().getText() || "";
  DriveApp.getFileById(docId).setTrashed(true);
  return text;
}


// Levenshtein distance for fuzzy hero matching
function levenshtein_(a, b) {
  const m = a.length, n = b.length;
  const dp = Array.from({length: m+1}, () => Array(n+1).fill(0));
  for (let i = 0; i <= m; i++) dp[i][0] = i;
  for (let j = 0; j <= n; j++) dp[0][j] = j;
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      dp[i][j] = a[i-1] === b[j-1]
        ? dp[i-1][j-1]
        : 1 + Math.min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1]);
    }
  }
  return dp[m][n];
}

function fuzzyMatchHero_(word) {
  if (!word || word.length < 3) return "";
  const w = word.toLowerCase();
  for (let i = 0; i < HEROES.length; i++) {
    if (w === HEROES[i].toLowerCase()) return HEROES[i];
  }
  let bestHero = "";
  let bestDist = 999;
  for (let i = 0; i < HEROES.length; i++) {
    const h = HEROES[i].toLowerCase();
    const maxDist = h.length >= 6 ? 2 : 1;
    const dist = levenshtein_(w, h);
    if (dist <= maxDist && dist < bestDist) {
      bestDist = dist;
      bestHero = HEROES[i];
    }
  }
  return bestHero;
}

function parsePostgameText_(text) {
  const src = String(text || "");
  const lines = src.split(/\r?\n/).map(s => s.trim()).filter(Boolean);
  const lower = src.toLowerCase();

  let mode = "";
  const modeMatch = lower.match(/\b(ranked|classic|custom|mro|tour|mcl)\b/i);
  if (modeMatch) {
    const m = modeMatch[1].toLowerCase();
    mode = m.charAt(0).toUpperCase() + m.slice(1);
  }

  let result = "";
  if (/\bwin|victory\b/i.test(lower)) result = "Win";
  else if (/\blose|loss|defeat\b/i.test(lower)) result = "Lose";

  let duration = 0;
  const durMin = lower.match(/\b(\d{1,2})\s*(m|min|minute|minutes)\b/i);
  if (durMin) duration = Number(durMin[1]) || 0;
  if (!duration) {
    const durClock = lower.match(/\b(\d{1,2})[:.](\d{2})\b/);
    if (durClock) duration = Number(durClock[1]) || 0;
  }

  const pairs = [];
  const usedHeroes = {};
  const heroesByLength = HEROES.slice().sort((a, b) => b.length - a.length);

  lines.forEach(line => {
    if (pairs.length >= 5) return;
    const lineLower = line.toLowerCase();
    let heroFound = "";
    for (let i = 0; i < heroesByLength.length; i++) {
      const h = heroesByLength[i];
      if (lineLower.indexOf(h.toLowerCase()) !== -1) {
        heroFound = h;
        break;
      }
    }
    // Pass 2: Fuzzy match - split line into words and try each
    if (!heroFound) {
      const words = line.split(/[\s|:;\-\[\]().,]+/).filter(w => w.length >= 3);
      for (let w = 0; w < words.length; w++) {
        const match = fuzzyMatchHero_(words[w]);
        if (match && !usedHeroes[match]) {
          heroFound = match;
          break;
        }
        if (w + 1 < words.length) {
          const combo = words[w] + " " + words[w+1];
          const match2 = fuzzyMatchHero_(combo);
          if (match2 && !usedHeroes[match2]) {
            heroFound = match2;
            break;
          }
        }
      }
    }

    if (!heroFound || usedHeroes[heroFound]) return;

    usedHeroes[heroFound] = true;
    const rx = new RegExp(heroFound.replace(/[.*+?^${}()|[\]\\]/g, "\\$&"), "ig");
    let player = line.replace(rx, " ").replace(/[|:\-\[\]()]/g, " ").replace(/\s+/g, " ").trim();
    if (!player) player = "Player " + (pairs.length + 1);

    pairs.push({ player: player, hero: heroFound });
  });

  return {
    date: new Date(),
    mode: mode,
    duration: duration,
    result: result,
    notes: "Auto OCR import",
    pairs: pairs,
    rawText: src
  };
}

function appendParsedGameToLog_(parsed) {
  const ss = getSafeActiveSpreadsheet();
  const sh = ss.getSheetByName("🎮 Game Log");
  const rows = sh.getRange(4, 1, 200, 16).getValues();
  let idx = -1;

  for (let i = 0; i < rows.length; i++) {
    const hasResult = rows[i][14] === "Win" || rows[i][14] === "Lose";
    const hasCore = rows[i][1] || rows[i][4] || rows[i][5];
    if (!hasResult && !hasCore) {
      idx = i;
      break;
    }
  }
  if (idx < 0) throw new Error("Game Log is full (200 rows).");

  const rowNum = idx + 4;
  const pairData = parsed.pairs.slice(0, 5);
  while (pairData.length < 5) pairData.push({ player: "", hero: "" });

  const seq = idx + 1;
  const values = [[
    seq,
    parsed.date || new Date(),
    parsed.mode || "",
    parsed.duration || "",
    pairData[0].player, pairData[0].hero,
    pairData[1].player, pairData[1].hero,
    pairData[2].player, pairData[2].hero,
    pairData[3].player, pairData[3].hero,
    pairData[4].player, pairData[4].hero,
    parsed.result,
    parsed.notes || "Auto OCR import"
  ]];

  sh.getRange(rowNum, 1, 1, 16).setValues(values);
  return rowNum;
}

function sortGameLogSheet(mode, direction) {
  const ss = getSafeActiveSpreadsheet();
  const sh = ss.getSheetByName("🎮 Game Log");
  const range = sh.getRange(4, 1, 200, 16);
  const rows = range.getValues();
  const filled = [];
  const empty = [];

  rows.forEach(row => {
    const hasData = row.some((value, index) => index !== 0 && value !== "" && value !== null);
    if (hasData) filled.push(row);
    else empty.push(row);
  });

  const resultRank = { Win: 1, Lose: 2 };
  const isDesc = direction === "desc";
  
  filled.sort((a, b) => {
    let cmp = 0;
    if (mode === "date") {
      cmp = new Date(a[1] || 0) - new Date(b[1] || 0);
    } else if (mode === "duration") {
      cmp = (Number(b[3]) || 0) - (Number(a[3]) || 0);
    } else if (mode === "result") {
      cmp = (resultRank[a[14]] || 99) - (resultRank[b[14]] || 99) || new Date(a[1] || 0) - new Date(b[1] || 0);
    }
    return isDesc ? -cmp : cmp;
  });

  const sortedRows = filled.concat(empty);
  sortedRows.forEach((row, index) => {
    row[0] = index < filled.length ? index + 1 : "";
  });

  range.setValues(sortedRows);
}
function readGameLog(ss) {
  const sh = ss.getSheetByName("🎮 Game Log");
  const data = sh.getRange(4, 1, 200, 16).getValues();
  const games = [];
  
  for (let i = 0; i < data.length; i++) {
    const row = data[i];
    const result = row[14];
    if (result !== "Win" && result !== "Lose") continue;
    
    const players = [];
    const heroes = [];
    for (let j = 0; j < 5; j++) {
      const p = row[4 + j*2];
      const h = row[5 + j*2];
      if (p && h) { players.push(p); heroes.push(h); }
    }
    
    if (players.length === 0) continue;
    
    games.push({
      num: row[0],
      date: row[1],
      mode: row[2],
      duration: row[3],
      players: players,
      heroes: heroes,
      result: result,
      notes: row[15],
      row: i + 4
    });
  }
  return games;
}

  const stats = {};
  games.forEach(g => {
    g.players.forEach((p, i) => {
      if (!stats[p]) stats[p] = { games: 0, wins: 0, heroes: {}, durations: [] };
      stats[p].games++;
      if (g.result === "Win") stats[p].wins++;
      if (g.duration > 0) stats[p].durations.push(g.duration);
      const h = g.heroes[i];
      if (h) {
        if (!stats[p].heroes[h]) stats[p].heroes[h] = { picks: 0, wins: 0 };
        stats[p].heroes[h].picks++;
        if (g.result === "Win") stats[p].heroes[h].wins++;
      }
    });
  });
  
  // Build sorted player list (by win rate descending, then games descending)
  const playerNames = Object.keys(stats);
  const playersSorted = playerNames.map((p, i) => {
    const s = stats[p];
    const wr = s.games > 0 ? (s.wins / s.games) * 100 : 0;
    const avgTime = s.durations.length > 0 ? (s.durations.reduce((a, b) => a + b, 0) / s.durations.length) : 0;
    return { name: p, idx: i, stats: s, wr: wr, avgTime: avgTime };
  }).sort((a, b) => {
    if (sortMode === "games") return compareDesc(a.stats.games, b.stats.games) || compareDesc(a.wr, b.wr);
    if (sortMode === "wins") return compareDesc(a.stats.wins, b.stats.wins) || compareDesc(a.stats.games, b.stats.games);
    if (sortMode === "losses") return compareDesc((a.stats.games - a.stats.wins), (b.stats.games - b.stats.wins)) || compareDesc(a.wr, b.wr);
    if (sortMode === "avgTime") return compareDesc(a.avgTime, b.avgTime) || compareDesc(a.stats.games, b.stats.games);
    return compareDesc(a.wr, b.wr) || compareDesc(a.stats.games, b.stats.games);
  });
  
  // Fill player rows (sorted by WR)
  playersSorted.forEach((pObj, rank) => {
    const r = 4 + rank;
    const p = pObj.name;
    const pi = pObj.idx;
    const s = pObj.stats;
    const wr = s.games > 0 ? Math.round((s.wins / s.games) * 100) : 0;
    const losses = s.games - s.wins;
    
    // Most picked hero
    let mostPicked = "-", mostPickedCount = 0;
    let bestHero = "-", bestHeroWR = 0;
    Object.entries(s.heroes).forEach(([hero, data]) => {
      if (data.picks > mostPickedCount) { mostPicked = hero; mostPickedCount = data.picks; }
      if (data.picks >= 2) {
        const hwr = (data.wins / data.picks) * 100;
        if (hwr > bestHeroWR || (hwr === bestHeroWR && data.picks > (s.heroes[bestHero]?.picks || 0))) {
          bestHero = hero; bestHeroWR = Math.round(hwr);
        }
      }
    });
    if (bestHero === "-" && Object.keys(s.heroes).length > 0) {
      const entries = Object.entries(s.heroes);
      entries.sort((a,b) => (b[1].wins/b[1].picks) - (a[1].wins/a[1].picks));
      bestHero = entries[0][0];
      bestHeroWR = Math.round((entries[0][1].wins / entries[0][1].picks) * 100);
    }
    
    // Win bar (visual)
    const barLen = 20;
    const filled = Math.round((wr / 100) * barLen);
    const winBar = "🟩".repeat(filled) + "⬛".repeat(barLen - filled) + " " + wr + "%";
    
    sh.getRange(r, 1).setValue(p);
    sh.getRange(r, 1).setFontWeight("bold").setFontColor(PLAYER_COLORS[pi % PLAYER_COLORS.length] || C.text);
    sh.getRange(r, 1, 1, 10)
      .setBackground(rank % 2 === 0 ? C.bg2 : C.bg)
      .setFontColor(C.text).setFontFamily("Arial").setFontSize(11)
      .setVerticalAlignment("middle");
    sh.getRange(r, 1).setFontWeight("bold").setFontColor(PLAYER_COLORS[pi % PLAYER_COLORS.length] || C.text);
    sh.getRange(r, 2).setValue(s.games);
    sh.getRange(r, 3).setValue(s.wins);
    sh.getRange(r, 4).setValue(losses);
    sh.getRange(r, 5).setValue(wr + "%");
    sh.getRange(r, 6).setValue(winBar);
    sh.getRange(r, 7).setValue(mostPicked);
    sh.getRange(r, 8).setValue(mostPickedCount);
    sh.getRange(r, 9).setValue(bestHero);
    sh.getRange(r, 10).setValue(bestHeroWR + "%");
    
    // Color the WR cell
    const wrColor = wr >= 60 ? "#1a4731" : wr >= 50 ? "#2d3a1a" : "#4a1a1a";
    const wrFont = wr >= 60 ? "#2ecc71" : wr >= 50 ? "#f1c40f" : "#e74c3c";
    sh.getRange(r, 5).setBackground(wrColor).setFontColor(wrFont).setFontWeight("bold");
    
    // Avg time
    const avgTime = s.durations.length > 0 ? Math.round(s.durations.reduce((a,b) => a+b, 0) / s.durations.length) : 0;
    sh.getRange(r, 11).setValue(avgTime > 0 ? avgTime + " min" : "-");
    sh.getRange(r, 11).setFontColor("#1abc9c");
  });
  
  // Per-player hero breakdown below
  let startRow = 210;
  
  playersSorted.forEach((pObj, pi) => {
    const p = pObj.name;
    const s = stats[p];
    const heroList = Object.entries(s.heroes).sort((a,b) => {
      const wrA = a[1].wins / a[1].picks, wrB = b[1].wins / b[1].picks;
      if (wrB !== wrA) return wrB - wrA; // sort by WR first
      return b[1].picks - a[1].picks; // then by picks
    });
    if (heroList.length === 0) return;
    
    // Player name header
    sh.getRange(startRow, 1, 1, 6).merge()
      .setValue("🎮 " + p.toUpperCase() + " — Hero Breakdown")
      .setBackground(PLAYER_COLORS[pi % PLAYER_COLORS.length] || "#333")
      .setFontColor("#fff").setFontWeight("bold").setFontSize(11);
    startRow++;
    
    sh.getRange(startRow, 1, 1, 6)
      .setValues([["Hero", "Role", "Picks", "Wins", "Losses", "Win %"]])
      .setBackground("#1a1a2e").setFontColor("#aaa").setFontWeight("bold").setFontSize(9);
    startRow++;
    
    heroList.forEach(([hero, data]) => {
      const wr = Math.round((data.wins / data.picks) * 100);
      const role = HERO_ROLES[hero] || "?";
      sh.getRange(startRow, 1, 1, 6).setValues([[hero, role, data.picks, data.wins, data.picks - data.wins, wr + "%"]]);
      sh.getRange(startRow, 1, 1, 6).setBackground("#0f0f23").setFontColor("#ecf0f1").setFontSize(10);
      
      const wrCol = wr >= 60 ? "#2ecc71" : wr >= 50 ? "#f1c40f" : "#e74c3c";
      sh.getRange(startRow, 6).setFontColor(wrCol).setFontWeight("bold");
      startRow++;
    });
    startRow++; // gap
  });
}

function updateHeroPool(ss, games) {
  const sh = ss.getSheetByName("🦸 Hero Pool");
  
  // Clear old data
  const lastRow = sh.getLastRow();
  if (lastRow > 3) sh.getRange(4, 1, lastRow - 3, 7).clear();
  
  // Aggregate hero stats
  const heroStats = {};
  games.forEach(g => {
    g.heroes.forEach(h => {
      if (!h) return;
      if (!heroStats[h]) heroStats[h] = { picks: 0, wins: 0 };
      heroStats[h].picks++;
      if (g.result === "Win") heroStats[h].wins++;
    });
  });
  
  const sortMode = getDocSetting("heroPoolSort", "picks");
  const sorted = Object.entries(heroStats).sort((a, b) => {
    const roleA = HERO_ROLES[a[0]] || "?";
    const roleB = HERO_ROLES[b[0]] || "?";
    const wrA = a[1].picks > 0 ? (a[1].wins / a[1].picks) * 100 : 0;
    const wrB = b[1].picks > 0 ? (b[1].wins / b[1].picks) * 100 : 0;
    const lossesA = a[1].picks - a[1].wins;
    const lossesB = b[1].picks - b[1].wins;

    if (sortMode === "wins") return compareDesc(a[1].wins, b[1].wins) || compareDesc(a[1].picks, b[1].picks);
    if (sortMode === "losses") return compareDesc(lossesA, lossesB) || compareDesc(a[1].picks, b[1].picks);
    if (sortMode === "wr") return compareDesc(wrA, wrB) || compareDesc(a[1].picks, b[1].picks);
    if (sortMode === "role") return getRoleRank(roleA) - getRoleRank(roleB) || compareDesc(a[1].picks, b[1].picks) || a[0].localeCompare(b[0]);
    return compareDesc(a[1].picks, b[1].picks) || compareDesc(a[1].wins, b[1].wins);
  });
  
  sorted.forEach(([hero, data], i) => {
    const r = 4 + i;
    const wr = Math.round((data.wins / data.picks) * 100);
    const losses = data.picks - data.wins;
    const role = HERO_ROLES[hero] || "?";
    
    // Pick bar
    const maxPicks = sorted[0][1].picks;
    const barLen = 15;
    const filled = Math.round((data.picks / maxPicks) * barLen);
    const pickBar = "🟦".repeat(filled) + "⬛".repeat(barLen - filled) + " " + data.picks;
    
    sh.getRange(r, 1, 1, 7).setValues([[hero, role, data.picks, data.wins, losses, wr + "%", pickBar]]);
    
    const rowBg = i % 2 === 0 ? "#1a1a2e" : "#0f0f23";
    sh.getRange(r, 1, 1, 7).setBackground(rowBg).setFontColor("#ecf0f1")
      .setFontFamily("Arial").setFontSize(10);
    
    // Role color
    const rc = getRoleColor(role);
    sh.getRange(r, 2).setBackground(rc.bg).setFontColor(rc.text).setFontWeight("bold");
    
    // WR color
    const wrCol = wr >= 60 ? "#2ecc71" : wr >= 50 ? "#f1c40f" : "#e74c3c";
    sh.getRange(r, 6).setFontColor(wrCol).setFontWeight("bold");
  });
}

function getRoleColor(role) {
  const map = {
    Tank:      { bg: "#1e3a5f", text: "#6db3f8" },
    Fighter:   { bg: "#5f3a1e", text: "#f8a86d" },
    Assassin:  { bg: "#4a1e5f", text: "#c86df8" },
    Mage:      { bg: "#1e5f5f", text: "#6df8f8" },
    Marksman:  { bg: "#5f5f1e", text: "#f8f86d" },
    Support:   { bg: "#1e5f3a", text: "#6df8a8" }
  };
  return map[role] || { bg: "#333", text: "#fff" };
}

function updateTeamComps(ss, games) {
  const sh = ss.getSheetByName("🧩 Team Comps");
  const lastRow = sh.getLastRow();
  if (lastRow > 2) sh.getRange(3, 1, lastRow - 2, 9).clear();
  
  // Add Notes column width
  sh.setColumnWidth(9, 280);
  
  // Separate win/lose compositions, collect notes for loses
  const comps = { Win: {}, Lose: {} };
  
  games.forEach(g => {
    if (g.heroes.length !== 5) return;
    const key = [...g.heroes].sort().join(" | ");
    if (!comps[g.result][key]) comps[g.result][key] = { heroes: [...g.heroes].sort(), count: 0, players: [], notes: [] };
    comps[g.result][key].count++;
    comps[g.result][key].players.push(g.players.join(", "));
    if (g.notes) comps[g.result][key].notes.push(g.notes);
  });
  
  let r = 3;
  
  // WIN COMPOSITIONS (sorted by count desc)
  sh.getRange(r, 1, 1, 9).merge()
    .setValue("🏆 WINNING COMPOSITIONS")
    .setBackground("#1a4731").setFontColor("#2ecc71").setFontWeight("bold")
    .setFontSize(14).setHorizontalAlignment("center");
  sh.setRowHeight(r, 40);
  r++;
  
  sh.getRange(r, 1, 1, 9)
    .setValues([["#","Hero 1","Hero 2","Hero 3","Hero 4","Hero 5","Times","Players","📝 Notes"]])
    .setBackground("#0d3320").setFontColor("#2ecc71").setFontWeight("bold").setFontSize(9);
  r++;
  
  const winComps = Object.values(comps.Win).sort((a,b) => b.count - a.count);
  if (winComps.length === 0) {
    sh.getRange(r, 1, 1, 9).merge().setValue("No winning compositions yet")
      .setBackground("#0f0f23").setFontColor("#7f8c8d").setHorizontalAlignment("center");

      sh.getRange(r, 1, 1, 9).setValues([vals]);
      sh.getRange(r, 1, 1, 9).setBackground(i%2===0 ? "#0f1a14" : "#0f0f23")
        .setFontColor("#ecf0f1").setFontSize(10);
      sh.getRange(r, 7).setFontColor("#2ecc71").setFontWeight("bold");
      sh.getRange(r, 9).setFontColor("#f39c12").setFontStyle("italic");
      r++;
    });
  }
  
  r++; // gap
  
  // LOSE COMPOSITIONS (sorted by count desc) — with Notes
  sh.getRange(r, 1, 1, 9).merge()
    .setValue("💀 LOSING COMPOSITIONS")
    .setBackground("#4a1a1a").setFontColor("#e74c3c").setFontWeight("bold")
    .setFontSize(14).setHorizontalAlignment("center");
  sh.setRowHeight(r, 40);
  r++;
  
  sh.getRange(r, 1, 1, 9)
    .setValues([["#","Hero 1","Hero 2","Hero 3","Hero 4","Hero 5","Times","Players","📝 Notes"]])
    .setBackground("#3d0d0d").setFontColor("#e74c3c").setFontWeight("bold").setFontSize(9);
  r++;
  
  const loseComps = Object.values(comps.Lose).sort((a,b) => b.count - a.count);
  if (loseComps.length === 0) {
    sh.getRange(r, 1, 1, 9).merge().setValue("No losing compositions yet")
      .setBackground("#0f0f23").setFontColor("#7f8c8d").setHorizontalAlignment("center");
    r++;

    loseComps.forEach((c, i) => {

      sh.getRange(r, 1, 1, 9).setBackground(i%2===0 ? "#1a0f0f" : "#0f0f23")
        .setFontColor("#ecf0f1").setFontSize(10);
      sh.getRange(r, 7).setFontColor("#e74c3c").setFontWeight("bold");
      sh.getRange(r, 9).setFontColor("#f39c12").setFontStyle("italic"); // Notes in orange italic
      r++;
    });
  }
}

function updateAnalytics(ss, games) {
  const sh = ss.getSheetByName("📊 Analytics");
  const lastRow = sh.getLastRow();
  if (lastRow > 2) sh.getRange(3, 1, lastRow - 2, 6).clear();
  
  let r = 3;
  
  // Overall Stats
  const totalW = games.filter(g => g.result === "Win").length;
  const totalL = games.filter(g => g.result === "Lose").length;
  const overallWR = games.length > 0 ? Math.round((totalW / games.length) * 100) : 0;
  
  // Streaks
  let bestWin = 0, worstLose = 0, curWin = 0, curLose = 0, curStreak = "";
  games.forEach(g => {
    if (g.result === "Win") { curWin++; curLose = 0; if (curWin > bestWin) bestWin = curWin; }
    else { curLose++; curWin = 0; if (curLose > worstLose) worstLose = curLose; }
  });
  if (games.length > 0) {
    const last = games[games.length-1].result;
    curStreak = last === "Win" ? curWin + "W" : curLose + "L";
  }
  
  // Summary cards
  const summaryData = [
    ["📊 OVERVIEW", "", "", "", "", ""],
    ["Total Games", games.length, "", "Total Wins", totalW, ""],
    ["Total Losses", totalL, "", "Overall WR", overallWR + "%", ""],
    ["Best Win Streak", bestWin + " 🔥", "", "Worst Lose Streak", worstLose + " ❄️", ""],
    ["Current Streak", curStreak, "", "", "", ""],
    ["", "", "", "", "", ""]
  ];
  
  sh.getRange(r, 1, 1, 6).merge()
    .setValue("📊 OVERVIEW").setBackground("#e94560").setFontColor("#fff")
    .setFontWeight("bold").setFontSize(14).setHorizontalAlignment("center");
  sh.setRowHeight(r, 40); r++;
  
  summaryData.slice(1).forEach(row => {
    sh.getRange(r, 1, 1, 6).setValues([row])
      .setBackground("#1a1a2e").setFontColor("#ecf0f1").setFontSize(11);
    sh.getRange(r, 1).setFontWeight("bold").setFontColor("#f1c40f");
    sh.getRange(r, 4).setFontWeight("bold").setFontColor("#f1c40f");
    r++;
  });
  
  r++;
  
  // Win Rate by Mode
  sh.getRange(r, 1, 1, 6).merge()
    .setValue("🎮 WIN RATE BY GAME MODE").setBackground("#0f3460").setFontColor("#fff")
    .setFontWeight("bold").setFontSize(14).setHorizontalAlignment("center");
  sh.setRowHeight(r, 40); r++;
  
  sh.getRange(r, 1, 1, 4).setValues([["Mode","Games","Wins","Win %"]])
    .setBackground("#0a2540").setFontColor("#3498db").setFontWeight("bold"); r++;
  
  const modes = {};
  games.forEach(g => {
    const m = g.mode || "Unknown";
    if (!modes[m]) modes[m] = { games: 0, wins: 0 };
    modes[m].games++;
    if (g.result === "Win") modes[m].wins++;
  });
  
  Object.entries(modes).forEach(([mode, data]) => {
    const wr = Math.round((data.wins / data.games) * 100);
    sh.getRange(r, 1, 1, 4).setValues([[mode, data.games, data.wins, wr + "%"]])
      .setBackground("#0f0f23").setFontColor("#ecf0f1").setFontSize(10);
    const wrCol = wr >= 60 ? "#2ecc71" : wr >= 50 ? "#f1c40f" : "#e74c3c";
    sh.getRange(r, 4).setFontColor(wrCol).setFontWeight("bold");
    r++;
  });
  
  r += 2;
  
  // Average Duration
  const durations = games.filter(g => g.duration > 0).map(g => g.duration);
  const avgDur = durations.length > 0 ? Math.round(durations.reduce((a,b)=>a+b,0) / durations.length) : 0;
  sh.getRange(r, 1, 1, 2).setValues([["⏱️ Avg Game Duration", avgDur > 0 ? avgDur + " min" : "No data"]])
    .setBackground("#1a1a2e").setFontColor("#ecf0f1").setFontSize(12);
  sh.getRange(r, 1).setFontWeight("bold").setFontColor("#f1c40f");
  r += 2;
  
  // Most played / best WR player
  const pStats = {};
  PLAYERS.forEach(p => pStats[p] = { games: 0, wins: 0 });
  games.forEach(g => g.players.forEach(p => {
    if (!pStats[p]) pStats[p] = { games: 0, wins: 0 };
    pStats[p].games++;
    if (g.result === "Win") pStats[p].wins++;
  }));
  
  const mostPlayed = Object.entries(pStats).sort((a,b) => b[1].games - a[1].games)[0];
  const bestWR = Object.entries(pStats).filter(e => e[1].games >= 3).sort((a,b) => (b[1].wins/b[1].games) - (a[1].wins/a[1].games))[0];
  
  sh.getRange(r, 1, 1, 2).setValues([["⭐ Most Games Played", mostPlayed ? mostPlayed[0] + " (" + mostPlayed[1].games + ")" : "-"]])
    .setBackground("#1a1a2e").setFontColor("#ecf0f1").setFontSize(12);
  sh.getRange(r, 1).setFontWeight("bold").setFontColor("#f1c40f");
  r++;
  
  sh.getRange(r, 1, 1, 2).setValues([["🎯 Highest WR Player (3+)", bestWR ? bestWR[0] + " (" + Math.round((bestWR[1].wins/bestWR[1].games)*100) + "%)" : "-"]])
    .setBackground("#1a1a2e").setFontColor("#ecf0f1").setFontSize(12);
  sh.getRange(r, 1).setFontWeight("bold").setFontColor("#f1c40f");
  r += 2;
  
  // Most picked hero overall
  const heroCount = {};
  games.forEach(g => g.heroes.forEach(h => { if (h) heroCount[h] = (heroCount[h]||0) + 1; }));
  const topHero = Object.entries(heroCount).sort((a,b) => b[1] - a[1])[0];
  
  sh.getRange(r, 1, 1, 2).setValues([["🎭 Most Picked Hero", topHero ? topHero[0] + " (" + topHero[1] + "x)" : "-"]])
    .setBackground("#1a1a2e").setFontColor("#ecf0f1").setFontSize(12);
  sh.getRange(r, 1).setFontWeight("bold").setFontColor("#f1c40f");
  r += 2;
  
  // SYNERGY MATRIX
  sh.getRange(r, 1, 1, 6).merge()
    .setValue("🤝 PLAYER SYNERGY MATRIX (Win% when playing together)")
    .setBackground("#5f27cd").setFontColor("#fff")
    .setFontWeight("bold").setFontSize(14).setHorizontalAlignment("center");
  sh.setRowHeight(r, 40); r++;
  
  // Header row
  const synergyHeader = [""].concat(PLAYERS);
  sh.getRange(r, 1, 1, PLAYERS.length + 1).setValues([synergyHeader])
    .setBackground("#2d1a4e").setFontColor("#c56df8").setFontWeight("bold").setFontSize(8);
  r++;
  
  PLAYERS.forEach((p1, i) => {
    const rowData = [p1];
    PLAYERS.forEach((p2, j) => {
      if (i === j) { rowData.push("—"); return; }
      // Find games where both played
      const together = games.filter(g => g.players.includes(p1) && g.players.includes(p2));
      if (together.length === 0) { rowData.push("-"); return; }
      const wins = together.filter(g => g.result === "Win").length;
      const wr = Math.round((wins / together.length) * 100);
      rowData.push(wr + "% (" + together.length + ")");
    });
    sh.getRange(r, 1, 1, PLAYERS.length + 1).setValues([rowData])
      .setBackground(i%2===0 ? "#1a1a2e" : "#0f0f23")
      .setFontColor("#ecf0f1").setFontSize(9);
    sh.getRange(r, 1).setFontWeight("bold").setFontColor(PLAYER_COLORS[i] || "#fff");
    r++;
  });
  
  r += 2;
  
  // WIN/LOSS TIMELINE
  sh.getRange(r, 1, 1, 6).merge()
    .setValue("📈 WIN/LOSS TIMELINE (last 30 games)")
    .setBackground("#e67e22").setFontColor("#fff")
    .setFontWeight("bold").setFontSize(14).setHorizontalAlignment("center");
  sh.setRowHeight(r, 40); r++;
  
  const recent = games.slice(-30);
  const timeline = recent.map(g => g.result === "Win" ? "🟩" : "🟥").join("");
  sh.getRange(r, 1, 1, 6).merge().setValue(timeline)
    .setBackground("#0f0f23").setFontColor("#ecf0f1").setFontSize(14).setHorizontalAlignment("center");
  r++;
  
  const timelineLabels = recent.map(g => "#" + g.num).join("  ");
  sh.getRange(r, 1, 1, 6).merge().setValue(timelineLabels)
    .setBackground("#0f0f23").setFontColor("#7f8c8d").setFontSize(8).setHorizontalAlignment("center");
}

function clearAllData() {
  const ui = SpreadsheetApp.getUi();
  const resp = ui.alert("⚠️ WARNING", "This will delete ALL game data and reset the system. Are you sure?", ui.ButtonSet.YES_NO);
  if (resp === ui.Button.YES) {
    const ss = getSafeActiveSpreadsheet();
    
    // Clear all data sheets BEFORE calling setup (so setup has nothing to backup)
    var sheetNames = ["🎮 Game Log", "👥 Player Stats", "🦸 Hero Pool", "🧩 Team Comps", "📊 Analytics", "📥 OCR Inbox"];
    sheetNames.forEach(function(name) {
      var sh = ss.getSheetByName(name);
      if (sh) sh.clear();
    });
    
    // Now run setup — it will find no data to backup, so it just creates fresh headers
    setupFullSystem();
    ui.alert("✅ All data cleared and system reset!");
  }
}

function processCsvImport(csvContent) {
  try {
    const data = Utilities.parseCsv(csvContent);
    if (data.length < 2) return { success: false, message: "File CSV kosong atau tiada baris data." };
    
    const headers = data[0].map(h => h.trim().toLowerCase());
    let imported = 0;
    
    const modeIdx = headers.indexOf("mode");
    const resultIdx = headers.indexOf("result");
    const durationIdx = headers.indexOf("duration");
    const notesIdx = headers.indexOf("notes");
    
    const pairsIdx = [];
    for(let i=1; i<=5; i++) {
      let p = headers.indexOf("player " + i);
      if(p === -1) p = headers.indexOf("player" + i);
      let h = headers.indexOf("hero " + i);
      if(h === -1) h = headers.indexOf("hero" + i);
      pairsIdx.push({ p: p, h: h });
    }
    
    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (row.length < 2 || !row.join("").trim()) continue;
      
      const pairs = [];
      for(let j=0; j<5; j++) {
        const pIdx = pairsIdx[j].p;
        const hIdx = pairsIdx[j].h;
        pairs.push({
          player: (pIdx !== -1 && row[pIdx]) ? row[pIdx].trim() : "",
          hero: (hIdx !== -1 && row[hIdx]) ? row[hIdx].trim() : ""
        });
      }
      
      const parsed = {
        date: new Date(),
        mode: modeIdx !== -1 ? row[modeIdx] : "Ranked",
        result: resultIdx !== -1 ? row[resultIdx] : "Win",
        duration: durationIdx !== -1 ? parseInt(row[durationIdx]) || 0 : 0,
        notes: notesIdx !== -1 ? row[notesIdx] : "",
        pairs: pairs
      };
      
      appendParsedGameToLog_(parsed);
      imported++;
    }
    
    refreshAllStats(true);
    return { success: true, message: "Berjaya import " + imported + " games!" };
  } catch (e) {
    return { success: false, message: "Error: " + e.message };
  }
}

function getDashboardData() {
  var ss = getSafeActiveSpreadsheet();
  var games = readGameLog(ss);
  if (games.length === 0) {
    return { totalGames: 0, wins: 0, losses: 0, winRate: 0, currentStreak: 0, streakType: "-", bestPlayer: "-", mostPickedHero: "-", avgDuration: 0, totalPlayers: 0 };
  }
  
  var wins = games.filter(function(g){ return g.result === "Win"; }).length;
  var losses = games.length - wins;
  var wr = Math.round((wins / games.length) * 100);
  
  // Streak
  var streak = 0, streakType = "";
  for (var i = games.length - 1; i >= 0; i--) {
    if (i === games.length - 1) { streakType = games[i].result; streak = 1; }
    else if (games[i].result === streakType) streak++;
    else break;
  }
  
  // Best player
  var pStats = {};
  games.forEach(function(g) {
    g.players.forEach(function(p) {
      if (!pStats[p]) pStats[p] = { games: 0, wins: 0 };
      pStats[p].games++;
      if (g.result === "Win") pStats[p].wins++;
    });
  });
  var bestPlayer = "-";
  var bestWR = -1;
  Object.keys(pStats).forEach(function(p) {
    if (pStats[p].games >= 3) {
      var pwr = (pStats[p].wins / pStats[p].games) * 100;
      if (pwr > bestWR) { bestWR = pwr; bestPlayer = p + " (" + Math.round(pwr) + "%)"; }
    }
  });
  
  // Most picked hero
  var heroCount = {};
  games.forEach(function(g) { g.heroes.forEach(function(h) { if (h) heroCount[h] = (heroCount[h]||0) + 1; }); });
  var topHero = "-";
  var topCount = 0;
  Object.keys(heroCount).forEach(function(h) { if (heroCount[h] > topCount) { topCount = heroCount[h]; topHero = h + " (" + topCount + "x)"; } });
  
  // Avg duration
  var durs = games.filter(function(g){ return g.duration > 0; }).map(function(g){ return g.duration; });
  var avgDur = durs.length > 0 ? Math.round(durs.reduce(function(a,b){return a+b;},0) / durs.length) : 0;
  
  // Unique players
  var allPlayers = {};
  games.forEach(function(g) { g.players.forEach(function(p) { allPlayers[p] = true; }); });
  
  return {
    totalGames: games.length,
    wins: wins,
    losses: losses,
    winRate: wr,
    currentStreak: streak,
    streakType: streakType,
    bestPlayer: bestPlayer,
    mostPickedHero: topHero,
    avgDuration: avgDur,
    totalPlayers: Object.keys(allPlayers).length
  };
}

function getLeaderboardData() {
  var ss = getSafeActiveSpreadsheet();
  var games = readGameLog(ss);
  var stats = {};
  
  games.forEach(function(g) {
    g.players.forEach(function(p, i) {
      if (!stats[p]) stats[p] = { games: 0, wins: 0, heroes: {} };
      stats[p].games++;
      if (g.result === "Win") stats[p].wins++;
      var h = g.heroes[i];
      if (h) {
        if (!stats[p].heroes[h]) stats[p].heroes[h] = { picks: 0, wins: 0 };
        stats[p].heroes[h].picks++;
        if (g.result === "Win") stats[p].heroes[h].wins++;
      }
    });
  });
  
  var result = [];
  Object.keys(stats).forEach(function(name, idx) {
    var s = stats[name];
    var wr = s.games > 0 ? Math.round((s.wins / s.games) * 100) : 0;
    // Find most picked hero
    var topHero = "-", topPicks = 0;
    Object.keys(s.heroes).forEach(function(h) {
      if (s.heroes[h].picks > topPicks) { topPicks = s.heroes[h].picks; topHero = h; }
    });
    result.push({
      name: name,
      games: s.games,
      wins: s.wins,
      losses: s.games - s.wins,
      wr: wr,
      topHero: topHero,
      color: PLAYER_COLORS[idx % PLAYER_COLORS.length] || "#fff"
    });
  });
  
  result.sort(function(a, b) { return b.wr - a.wr || b.games - a.games; });
  return result;
}

function getDraftHelperData() {
  var ss = getSafeActiveSpreadsheet();
  var games = readGameLog(ss);
  var heroStats = {};
  
  games.forEach(function(g) {
    g.heroes.forEach(function(h) {
      if (!h) return;
      if (!heroStats[h]) heroStats[h] = { picks: 0, wins: 0 };
      heroStats[h].picks++;
      if (g.result === "Win") heroStats[h].wins++;
    });
  });
  
  var roles = { Tank: [], Fighter: [], Assassin: [], Mage: [], Marksman: [], Support: [] };
  Object.keys(heroStats).forEach(function(hero) {
    var role = HERO_ROLES[hero] || "?";
    if (!roles[role]) return;
    var s = heroStats[hero];
    if (s.picks < 2) return;
    roles[role].push({
      hero: hero,
      picks: s.picks,
      wins: s.wins,
      wr: Math.round((s.wins / s.picks) * 100)
    });
  });
  
  // Sort each role by WR desc, take top 3
  Object.keys(roles).forEach(function(role) {
    roles[role].sort(function(a, b) { return b.wr - a.wr || b.picks - a.picks; });
    roles[role] = roles[role].slice(0, 3);
  });
  
  return roles;
}

function getChartData() {
  var ss = getSafeActiveSpreadsheet();
  var games = readGameLog(ss);
  
  // Timeline (last 30)
  var timeline = games.slice(-30).map(function(g) {
    return { num: g.num, result: g.result };
  });
  
  // Role distribution
  var roleDist = { Tank: 0, Fighter: 0, Assassin: 0, Mage: 0, Marksman: 0, Support: 0 };
  games.forEach(function(g) {
    g.heroes.forEach(function(h) {
      var role = HERO_ROLES[h] || "?";
      if (roleDist[role] !== undefined) roleDist[role]++;
    });
  });
  
  // Mode distribution
  var modeDist = {};
  games.forEach(function(g) {
    var m = g.mode || "Unknown";
    modeDist[m] = (modeDist[m] || 0) + 1;
  });
  
  return { timeline: timeline, roleDist: roleDist, modeDist: modeDist };
}

function getAllWebAppData() {
  return {
    dashboard: getDashboardData(),
    leaderboard: getLeaderboardData(),
    draft: getDraftHelperData(),
    charts: getChartData()
  };
}
