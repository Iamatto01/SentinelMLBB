var GLOBAL_SS = null;

// Helper function to allow Code.gs and Code_Part2.gs to seamlessly work with different sheets
function getSafeActiveSpreadsheet() {
  if (typeof GLOBAL_SS !== 'undefined' && GLOBAL_SS) return GLOBAL_SS;
  return SpreadsheetApp.getActiveSpreadsheet();
}

function doGet(e) {
  var action = e.parameter.action;
  var email = e.parameter.email;
  var sheetId = e.parameter.sheetId;
  
  // Set the global SS if sheetId is provided (for multi-tenant data fetching)
  if (sheetId) {
    try {
      GLOBAL_SS = SpreadsheetApp.openById(sheetId);
    } catch (err) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Invalid Sheet ID or no permission" })).setMimeType(ContentService.MimeType.JSON);
    }
  }
  
  try {
    if (action === "login") {
      // Login reads from the Admin's Master Sheet where this script is bound
      var masterSheet = SpreadsheetApp.getActiveSpreadsheet();
      var usersTab = masterSheet.getSheetByName("SaaS_Users");
      
      // Auto-create SaaS_Users tab if it doesn't exist
      if (!usersTab) {
        usersTab = masterSheet.insertSheet("SaaS_Users");
        usersTab.appendRow(["Email", "SheetID", "Status"]);
        usersTab.getRange("A1:C1").setFontWeight("bold");
      }
      
      var data = usersTab.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        if (data[i][0].toString().toLowerCase() === email.toLowerCase() && data[i][2] === "Active") {
          return ContentService.createTextOutput(JSON.stringify({ 
            status: "success", 
            sheetId: data[i][1],
            role: email.toLowerCase() === "admin@sentinelmlbb.com" ? "admin" : "user"
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "User not found or inactive." })).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (action === "getGames") {
      var games = readGameLog(GLOBAL_SS || SpreadsheetApp.getActiveSpreadsheet()); // Use GLOBAL_SS
      return ContentService.createTextOutput(JSON.stringify({ status: "success", data: games }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Unknown action" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  var output = {};
  try {
    var payload = JSON.parse(e.postData.contents);
    var action = payload.action;
    var sheetId = payload.sheetId;
    
    if (sheetId) {
      GLOBAL_SS = SpreadsheetApp.openById(sheetId);
    }

    if (action === "addGame") {
      var parsed = payload.data;
      var insertedRow = appendParsedGameToLog_(parsed); // Uses getSafeActiveSpreadsheet internally now
      output = { status: "success", row: insertedRow };
    } else if (action === "updateGame") {
      var parsed = payload.data;
      var gameNum = payload.gameNum;
      var updatedRow = updateParsedGameInLog_(parsed, gameNum);
      output = { status: "success", row: updatedRow };
    } else if (action === "registerUser") {
      // Admin registering a new user — auto-creates a clean sheet for the client
      var masterSheet = SpreadsheetApp.getActiveSpreadsheet();
      var usersTab = masterSheet.getSheetByName("SaaS_Users");
      if (!usersTab) {
        usersTab = masterSheet.insertSheet("SaaS_Users");
        usersTab.appendRow(["Email", "SheetID", "Status", "Created"]);
        usersTab.getRange("A1:D1").setFontWeight("bold");
      }

      // Check if email already exists
      var existingData = usersTab.getDataRange().getValues();
      for (var i = 1; i < existingData.length; i++) {
        if (existingData[i][0].toString().toLowerCase() === payload.email.toLowerCase()) {
          output = { status: "error", message: "Email already registered." };
          return ContentService.createTextOutput(JSON.stringify(output)).setMimeType(ContentService.MimeType.JSON);
        }
      }

      // Create a brand new spreadsheet for the client (NO scripts attached!)
      var clientName = payload.email.split("@")[0];
      var newSS = SpreadsheetApp.create("SentinelMLBB - " + clientName);
      var newSheetId = newSS.getId();

      // Setup all required tabs in the new sheet
      var tabs = ["🎮 Game Log", "👥 Player Stats", "🦸 Hero Pool", "🧩 Team Comps", "📊 Analytics", "📥 OCR Inbox"];
      tabs.forEach(function(name) {
        newSS.insertSheet(name);
      });
      // Delete the default "Sheet1"
      var defaultSheet = newSS.getSheetByName("Sheet1");
      if (defaultSheet && newSS.getSheets().length > 1) newSS.deleteSheet(defaultSheet);

      // Run setup functions on the new sheet
      GLOBAL_SS = newSS;
      try {
        setupGameLog(newSS);
        setupPlayerStats(newSS);
        setupHeroPool(newSS);
        setupTeamComps(newSS);
        setupAnalytics(newSS);
        setupOcrInbox(newSS);
      } catch(setupErr) {
        // Setup functions are optional, continue even if they fail
        Logger.log("Setup warning: " + setupErr.message);
      }
      GLOBAL_SS = null;

      // Share with client email (Editor access)
      try {
        newSS.addEditor(payload.email);
      } catch(shareErr) {
        Logger.log("Could not share with " + payload.email + ": " + shareErr.message);
      }

      // Register in SaaS_Users
      usersTab.appendRow([payload.email, newSheetId, "Active", new Date().toISOString()]);
      output = { status: "success", message: "Client registered! Sheet auto-created.", sheetId: newSheetId };
    } else {
      output = { status: "error", message: "Unknown action" };
    }
  } catch (err) {
    output = { status: "error", message: err.message };
  }
  
  return ContentService.createTextOutput(JSON.stringify(output))
    .setMimeType(ContentService.MimeType.JSON);
}
