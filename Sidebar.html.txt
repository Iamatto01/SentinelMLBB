  <script>
    // ============ TAB SWITCHING ============
    function switchTab(tabId, btn) {
      document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));
      document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));
      document.getElementById('tab-' + tabId).classList.add('active');
      btn.classList.add('active');
      hideStatus();
      // Auto-load games when switching to games tab
      if (tabId === 'games') loadGames();
    }

    // ============ STATUS HELPERS ============
    function showStatus(msg, isSuccess) {
      const el = document.getElementById('status');
      el.textContent = msg;
      el.className = 'status ' + (isSuccess ? 'success' : 'error');
      document.getElementById('loader').style.display = 'none';
    }
    function hideStatus() {
      document.getElementById('status').className = 'status';
      document.getElementById('loader').style.display = 'none';
    }
    function showLoader() {
      hideStatus();
      document.getElementById('loader').style.display = 'block';
    }

    // ============ CSV IMPORT ============
    function importCSV() {
      const fileInput = document.getElementById('csvFile');
      if (!fileInput.files.length) {
        showStatus('Sila pilih fail CSV terlebih dahulu.', false);
        return;
      }
      showLoader();
      document.getElementById('btnImport').disabled = true;
      const reader = new FileReader();
      reader.onload = function(e) {
        google.script.run
          .withSuccessHandler(function(res) {
            document.getElementById('btnImport').disabled = false;
            showStatus(res.message, res.success);
            if(res.success) fileInput.value = '';
          })
          .withFailureHandler(function(err) {
            document.getElementById('btnImport').disabled = false;
            showStatus(err.message || err, false);
          })
          .processCsvImport(e.target.result);
      };
      reader.readAsText(fileInput.files[0]);
    }

    // ============ LINEUP UI ============
    window.onload = function() {
      google.script.run.withSuccessHandler(function(players) {
        google.script.run.withSuccessHandler(function(heroes) {
          buildLineupUI(players, heroes);
        }).getHeroesList();
      }).getPlayersList();
    };

    function buildLineupUI(players, heroes) {
      const container = document.getElementById('lineupContainer');
      let html = '';
      const pOptions = '<option value="">Player...</option>' + 
        players.map(p => '<option value="'+p+'">'+p+'</option>').join('');
      for(let i=1; i<=5; i++) {
        html += '<div class="player-row">'
          + '<span class="slot">'+i+'</span>'
          + '<select name="player'+i+'">'+pOptions+'</select>'
          + '<input type="text" name="hero'+i+'" placeholder="Hero..." list="heroList" />'
          + '</div>';
      }
      html += '<datalist id="heroList">' + heroes.map(h => '<option value="'+h+'">').join('') + '</datalist>';
      container.innerHTML = html;
    }

    // ============ MANUAL ADD ============
    function addManualGame() {
      const form = document.getElementById('addGameForm');
      let hasPlayer = false;
      for(let i=1; i<=5; i++) {
        if(form['player'+i].value.trim() && form['hero'+i].value.trim()) { hasPlayer = true; break; }
      }
      if(!hasPlayer) { showStatus('Pilih sekurang-kurangnya 1 Player dan Hero!', false); return; }

      showLoader();
      document.getElementById('btnAdd').disabled = true;

      // Collect form data manually (google.script.run can't serialize form elements directly in sidebar)
      var formData = {
        mode: form.mode.value,
        result: form.result.value,
        duration: form.duration.value,
        notes: form.notes.value
      };
      for(var i=1; i<=5; i++) {
        formData['player'+i] = form['player'+i].value;
        formData['hero'+i] = form['hero'+i].value;
      }

      google.script.run
        .withSuccessHandler(function(res) {
          document.getElementById('btnAdd').disabled = false;
          showStatus(res.message, res.success);
          if(res.success) {
            // Reset only the lineup and notes, keep mode/result/duration
            form.notes.value = '';
            for(var i=1; i<=5; i++) { form['player'+i].value = ''; form['hero'+i].value = ''; }
          }
        })
        .withFailureHandler(function(err) {
          document.getElementById('btnAdd').disabled = false;
          showStatus(err.message || err, false);
        })
        .processManualAddGame(formData);
    }

    // ============ GAMES LIST / DELETE ============
    function loadGames() {
      var container = document.getElementById('gamesList');
      container.innerHTML = '<div class="empty-state">Loading...</div>';
      google.script.run
        .withSuccessHandler(function(games) {
          if (!games || games.length === 0) {
            container.innerHTML = '<div class="empty-state">Tiada game log dijumpai.</div>';
            return;
          }
          var html = '';
          games.forEach(function(g) {
            var badgeClass = g.result === 'Win' ? 'badge-win' : 'badge-lose';
            html += '<div class="game-card">'
              + '<span class="game-num">#'+g.num+'</span>'
              + '<div class="game-info">'
              + '<div class="top">'
              + '<span class="badge '+badgeClass+'">'+g.result+'</span>'
              + '<span class="badge badge-mode">'+g.mode+'</span>'
              + '<span style="color:var(--dim);font-size:11px;">'+g.date+'</span>'
              + '</div>'
              + '<div class="lineup">'+g.lineup+'</div>'
              + '</div>'
              + '<button class="delete-btn" onclick="deleteGame('+g.num+')" title="Delete">🗑</button>'
              + '</div>';
          });
          container.innerHTML = html;
        })
        .withFailureHandler(function(err) {
          container.innerHTML = '<div class="empty-state" style="color:var(--lose)">Error: '+(err.message||err)+'</div>';
        })
        .getGamesList();
    }

    function deleteGame(num) {
      if (!confirm('Padam Game #'+num+'?')) return;
      showLoader();
      google.script.run
        .withSuccessHandler(function(res) {
          showStatus(res.message, res.success);
          if(res.success) loadGames(); // Refresh list
        })
        .withFailureHandler(function(err) {
          showStatus(err.message || err, false);
        })
        .deleteGameFromSidebar(num);
    }

    // ============ RUN FUNCTION ============
    function runFunction(funcName) {
      showLoader();
      google.script.run
        .withSuccessHandler(function() { showStatus('Berjaya!', true); })
        .withFailureHandler(function(err) { showStatus(err.message || err, false); })
        [funcName]();
    }
  </script>
