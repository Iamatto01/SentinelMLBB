<script>
var _players=[],_heroes=[];
var roleColors={Tank:'#3b82f6',Fighter:'#f59e0b',Assassin:'#ef4444',Mage:'#8b5cf6',Marksman:'#06b6d4',Support:'#10b981'};
var roleIcons={Tank:'🛡',Fighter:'⚔',Assassin:'🗡',Mage:'🔮',Marksman:'🏹',Support:'💚'};

function go(id,btn){
  document.querySelectorAll('.page').forEach(function(p){p.classList.remove('active')});
  document.querySelectorAll('.nav button').forEach(function(b){b.classList.remove('active')});
  document.getElementById('pg-'+id).classList.add('active');
  btn.classList.add('active');
  if(id==='home')loadDashboard();
  if(id==='rank')loadLeaderboard();
  if(id==='draft')loadDraft();
  if(id==='games')loadGames();
}

function loadDashboard(){
  document.getElementById('dashLoader').style.display='block';
  document.getElementById('dashContent').style.display='none';
  google.script.run.withSuccessHandler(function(d){
    document.getElementById('dashLoader').style.display='none';
    document.getElementById('dashContent').style.display='block';
    renderDashboard(d);
  }).withFailureHandler(function(e){
    document.getElementById('dashLoader').innerHTML='<span style="color:var(--lose)">Error: '+e.message+'</span>';
  }).getAllWebAppData();
}

function renderDashboard(d){
  var db=d.dashboard,ch=d.charts;
  var g=document.getElementById('statsGrid');
  g.innerHTML=
    card(db.totalGames,'Total Games','accent')+
    card(db.winRate+'%','Win Rate',db.winRate>=50?'win':'lose')+
    card(db.wins,'Wins','win')+card(db.losses,'Losses','lose')+
    card(db.currentStreak+' '+db.streakType,'Current Streak','gold')+
    card(db.avgDuration+'m','Avg Duration','accent')+
    '<div class="stat-card wide accent"><div class="value" style="font-size:15px">⭐ '+db.bestPlayer+'</div><div class="label">Best Player (WR)</div></div>'+
    '<div class="stat-card wide gold"><div class="value" style="font-size:15px">🎭 '+db.mostPickedHero+'</div><div class="label">Most Picked Hero</div></div>';
  // Timeline
  var tl=document.getElementById('timeline');
  tl.innerHTML=ch.timeline.map(function(t){return '<div class="tl-bar '+(t.result==='Win'?'w':'l')+'" title="Game #'+t.num+' - '+t.result+'"></div>'}).join('');
  // Role bars
  var rb=document.getElementById('roleBars');
  var maxR=Math.max.apply(null,Object.values(ch.roleDist))||1;
  var html='';
  Object.keys(ch.roleDist).forEach(function(r){
    var v=ch.roleDist[r],pct=Math.round((v/maxR)*100);
    html+='<div class="role-bar-wrap"><div class="role-bar-label"><span>'+roleIcons[r]+' '+r+'</span><span>'+v+' picks</span></div><div class="role-bar"><div class="role-bar-fill" style="width:'+pct+'%;background:'+roleColors[r]+'"></div></div></div>';
  });
  rb.innerHTML=html;
}

function card(val,lbl,cls){return '<div class="stat-card '+cls+'"><div class="value">'+val+'</div><div class="label">'+lbl+'</div></div>'}

function loadLeaderboard(){
  document.getElementById('lbList').innerHTML='<div class="empty">Loading...</div>';
  google.script.run.withSuccessHandler(function(data){
    if(!data||data.length===0){document.getElementById('lbList').innerHTML='<div class="empty">Tiada data.</div>';return}
    var medals=['🥇','🥈','🥉'];
    var html=data.map(function(p,i){
      var wrColor=p.wr>=60?'var(--win)':p.wr>=45?'var(--gold)':'var(--lose)';
      return '<div class="lb-card"><div class="lb-rank">'+(medals[i]||'#'+(i+1))+'</div><div class="lb-info"><div class="lb-name" style="color:'+p.color+'">'+p.name+'</div><div class="lb-sub">'+p.topHero+' main · '+p.wins+'W '+p.losses+'L</div><div class="wr-bar"><div class="wr-bar-fill" style="width:'+p.wr+'%"></div></div></div><div class="lb-wr"><div class="pct" style="color:'+wrColor+'">'+p.wr+'%</div><div class="games">'+p.games+' games</div></div></div>'
    }).join('');
    document.getElementById('lbList').innerHTML=html;
  }).getLeaderboardData();
}

function loadDraft(){
  document.getElementById('draftContent').innerHTML='<div class="empty">Loading...</div>';
  google.script.run.withSuccessHandler(function(data){
    var html='';
    Object.keys(data).forEach(function(role){
      var heroes=data[role];
      var c=roleColors[role]||'#888';
      html+='<div class="draft-role"><div class="draft-role-title" style="background:'+c+'22;color:'+c+'">'+roleIcons[role]+' '+role+'</div><div class="draft-heroes">';
      if(heroes.length===0){html+='<div class="draft-hero"><span class="name" style="color:var(--dim)">No data (min 2 picks)</span></div>'}
      else{var med=['🥇','🥈','🥉'];heroes.forEach(function(h,i){
        var wc=h.wr>=60?'var(--win)':h.wr>=45?'var(--gold)':'var(--lose)';
        html+='<div class="draft-hero"><span class="medal">'+med[i]+'</span><span class="name">'+h.hero+'</span><span class="wr" style="color:'+wc+'">'+h.wr+'%</span><span class="picks">'+h.picks+' picks</span></div>'
      })}
      html+='</div></div>';
    });
    document.getElementById('draftContent').innerHTML=html;
  }).getDraftHelperData();
}

function loadGames(){
  document.getElementById('gamesList').innerHTML='<div class="empty">Loading...</div>';
  google.script.run.withSuccessHandler(function(games){
    if(!games||games.length===0){document.getElementById('gamesList').innerHTML='<div class="empty">Tiada game log.</div>';return}
    document.getElementById('gamesList').innerHTML=games.map(function(g){
      return '<div class="game-row"><span class="num">#'+g.num+'</span><div class="info"><div class="top"><span class="badge '+(g.result==='Win'?'badge-w':'badge-l')+'">'+g.result+'</span><span class="badge badge-m">'+g.mode+'</span><span style="color:var(--dim);font-size:11px">'+g.date+'</span></div><div class="lineup">'+g.lineup+'</div></div><button class="del-btn" onclick="delGame('+g.num+')">🗑</button></div>'
    }).join('');
  }).getGamesList();
}

function delGame(n){if(confirm('Padam Game #'+n+'?')){showGStatus('Deleting...',true);google.script.run.withSuccessHandler(function(r){showGStatus(r.message,r.success);if(r.success)loadGames()}).deleteGameFromSidebar(n)}}

function openAddModal(){
  document.getElementById('addModal').classList.add('show');
  if(_players.length===0){
    google.script.run.withSuccessHandler(function(p){_players=p;google.script.run.withSuccessHandler(function(h){_heroes=h;buildForm()}).getHeroesList()}).getPlayersList();
  }
}
function closeAddModal(){document.getElementById('addModal').classList.remove('show');document.getElementById('mStatus').className='status'}

function buildForm(){
  var c=document.getElementById('fLineup'),html='',opts='<option value="">Player...</option>';
  _players.forEach(function(p){opts+='<option value="'+p+'">'+p+'</option>'});
  for(var i=1;i<=5;i++){html+='<div class="player-row"><span class="slot">'+i+'</span><select id="fp'+i+'">'+opts+'</select><input type="text" id="fh'+i+'" placeholder="Hero..." list="hlst"></div>'}
  html+='<datalist id="hlst">'+_heroes.map(function(h){return '<option value="'+h+'">'}).join('')+'</datalist>';
  c.innerHTML=html;
}

function saveGame(){
  var fd={mode:document.getElementById('fMode').value,result:document.getElementById('fResult').value,duration:document.getElementById('fDur').value,notes:document.getElementById('fNotes').value};
  var has=false;
  for(var i=1;i<=5;i++){fd['player'+i]=document.getElementById('fp'+i).value;fd['hero'+i]=document.getElementById('fh'+i).value;if(fd['player'+i]&&fd['hero'+i])has=true}
  if(!has){showMStatus('Pilih min 1 Player+Hero!',false);return}
  document.getElementById('btnSave').disabled=true;
  google.script.run.withSuccessHandler(function(r){
    document.getElementById('btnSave').disabled=false;showMStatus(r.message,r.success);
    if(r.success){for(var i=1;i<=5;i++){document.getElementById('fp'+i).value='';document.getElementById('fh'+i).value=''}document.getElementById('fNotes').value='';setTimeout(function(){closeAddModal();loadGames()},1000)}
  }).withFailureHandler(function(e){document.getElementById('btnSave').disabled=false;showMStatus(e.message,false)}).processManualAddGame(fd);
}

function importCSV(){
  var fi=document.getElementById('csvFile');
  if(!fi.files.length){showGStatus('Pilih fail CSV dulu.',false);return}
  document.getElementById('btnCsv').disabled=true;
  var r=new FileReader();r.onload=function(e){
    google.script.run.withSuccessHandler(function(res){document.getElementById('btnCsv').disabled=false;showGStatus(res.message,res.success);if(res.success)fi.value=''}).withFailureHandler(function(err){document.getElementById('btnCsv').disabled=false;showGStatus(err.message,false)}).processCsvImport(e.target.result);
  };r.readAsText(fi.files[0]);
}

function doRefresh(){showGStatus('Refreshing...',true);google.script.run.withSuccessHandler(function(){showGStatus('Stats refreshed!',true)}).withFailureHandler(function(e){showGStatus(e.message,false)}).refreshAllStats(true)}
function doAction(fn){google.script.run.withSuccessHandler(function(){showGStatus('Done!',true)}).withFailureHandler(function(e){showGStatus(e.message,false)})[fn]()}
function showGStatus(m,ok){var e=document.getElementById('gStatus');e.textContent=m;e.className='status '+(ok?'ok':'err')}
function showMStatus(m,ok){var e=document.getElementById('mStatus');e.textContent=m;e.className='status '+(ok?'ok':'err')}

// Auto-load dashboard on start
window.onload=function(){loadDashboard()};
