        // Navigation
        function switchPage(pageId, btnElement) {
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
            document.querySelectorAll('.nav-item').forEach(b => b.classList.remove('active'));
            document.getElementById('pg-' + pageId).classList.add('active');
            if(btnElement) btnElement.classList.add('active');
            
            if(pageId === 'home') loadDashboard();
            if(pageId === 'rank') loadLeaderboard();
            if(pageId === 'draft') loadDraft();
            if(pageId === 'games') loadGames();
        }

        // Dashboard
        function loadDashboard() {
            document.getElementById('dashLoader').style.display = 'block';
            document.getElementById('dashContent').style.display = 'none';
            google.script.run.withSuccessHandler(res => {
                document.getElementById('dashLoader').style.display = 'none';
                document.getElementById('dashContent').style.display = 'block';
                renderDashboard(res);
            }).withFailureHandler(err => showToast(err.message, false)).getAllWebAppData();
        }

        function renderDashboard(data) {
            const db = data.dashboard;
            const ch = data.charts;
            
            // Cards
            document.getElementById('mainStats').innerHTML = `
                <div class="stat-box box-accent">
                    <i class="fa-solid fa-gamepad bg-icon"></i>
                    <div class="stat-value">${db.totalGames}</div>
                    <div class="stat-label">Total Matches</div>
                </div>
                <div class="stat-box ${db.winRate >= 50 ? 'box-win' : 'box-lose'}">
                    <i class="fa-solid fa-percent bg-icon"></i>
                    <div class="stat-value">${db.winRate}%</div>
                    <div class="stat-label">Win Rate</div>
                </div>
                <div class="stat-box box-win">
                    <i class="fa-solid fa-arrow-trend-up bg-icon"></i>
                    <div class="stat-value" style="color:var(--success)">${db.wins}</div>
                    <div class="stat-label">Victories</div>
                </div>
                <div class="stat-box box-lose">
                    <i class="fa-solid fa-arrow-trend-down bg-icon"></i>
                    <div class="stat-value" style="color:var(--danger)">${db.losses}</div>
                    <div class="stat-label">Defeats</div>
                </div>
                <div class="stat-box wide box-gold">
                    <i class="fa-solid fa-fire bg-icon"></i>
                    <div>
                        <div class="stat-label" style="margin-top:0">Current Streak</div>
                        <div class="stat-value">${db.currentStreak} ${db.streakType}</div>
                    </div>
                    <div style="text-align: right">
                        <div class="stat-label" style="margin-top:0">MVP (WR)</div>
                        <div class="stat-value" style="font-size:20px">${db.bestPlayer}</div>
                    </div>
                </div>
            `;

            // Timeline
            document.getElementById('timelineBar').innerHTML = ch.timeline.map(t => 
                `<div class="tl-bar ${t.result === 'Win' ? 'tl-w' : 'tl-l'}" title="Game #${t.num}: ${t.result}"></div>`
            ).join('');

            // Roles
            const maxR = Math.max(...Object.values(ch.roleDist)) || 1;
            let roleHtml = '';
            for (const [role, count] of Object.entries(ch.roleDist)) {
                if(count === 0) continue;
                const pct = Math.round((count / maxR) * 100);
                const color = roleColors[role] || '#fff';
                const icon = roleIcons[role] || 'fa-user';
                roleHtml += `
                    <div class="role-row">
                        <div class="role-labels">
                            <span><i class="fa-solid ${icon}" style="color:${color}"></i> ${role}</span>
                            <span style="color:var(--text-muted)">${count} picks</span>
                        </div>
                        <div class="role-track">
                            <div class="role-fill" style="width: ${pct}%; background: ${color}; box-shadow: 0 0 10px ${color}"></div>
                        </div>
                    </div>
                `;
            }
            document.getElementById('roleDistBars').innerHTML = roleHtml || '<div class="stat-label text-center">No data yet</div>';
        }

        // Leaderboard
        function loadLeaderboard() {
            document.getElementById('lbContent').innerHTML = '<div class="full-loader"><div class="spinner"></div></div>';
            google.script.run.withSuccessHandler(data => {
                if(!data.length) { document.getElementById('lbContent').innerHTML = '<div class="full-loader">No data</div>'; return; }
                
                let html = '';
                data.forEach((p, i) => {
                    let rankHtml = `<div class="rank-num">${i+1}</div>`;
                    if(i===0) rankHtml = `<div class="rank-num rank-1"><i class="fa-solid fa-crown"></i></div>`;
                    else if(i===1) rankHtml = `<div class="rank-num rank-2"><i class="fa-solid fa-medal"></i></div>`;
                    else if(i===2) rankHtml = `<div class="rank-num rank-3"><i class="fa-solid fa-award"></i></div>`;

                    const wrColor = p.wr >= 60 ? 'var(--success)' : p.wr >= 45 ? 'var(--warning)' : 'var(--danger)';

                    html += `
                        <div class="rank-card">
                            ${rankHtml}
                            <div class="rank-info">
                                <div class="rank-name" style="color: ${p.color}">${p.name}</div>
                                <div class="rank-stats"><i class="fa-solid fa-hand-fist"></i> ${p.topHero} main &bull; ${p.wins}W ${p.losses}L</div>
                            </div>
                            <div class="rank-wr-box">
                                <div class="rank-wr-val" style="color: ${wrColor}">${p.wr}%</div>
                                <div class="rank-wr-bar"><div class="rank-wr-fill" style="width: ${p.wr}%; background: ${wrColor}"></div></div>
                            </div>
                        </div>
                    `;
                });
                document.getElementById('lbContent').innerHTML = html;
            }).getLeaderboardData();
        }

        // Draft
        function loadDraft() {
            document.getElementById('draftContent').innerHTML = '<div class="full-loader"><div class="spinner"></div></div>';
            google.script.run.withSuccessHandler(data => {
                let html = '';
                for (const [role, heroes] of Object.entries(data)) {
                    const color = roleColors[role] || '#fff';
                    const icon = roleIcons[role] || 'fa-user';
                    
                    html += `
                        <div class="draft-role-card">
                            <div class="draft-header">
                                <i class="fa-solid ${icon}" style="color:${color}"></i> ${role}
                            </div>
                    `;
                    
                    if(heroes.length === 0) {
                        html += `<div class="draft-item"><span class="draft-name" style="color:var(--text-muted)">Need more data</span></div>`;
                    } else {
                        const medals = ['<i class="fa-solid fa-crown" style="color:#FFD700"></i>', '<i class="fa-solid fa-medal" style="color:#C0C0C0"></i>', '<i class="fa-solid fa-award" style="color:#CD7F32"></i>'];
                        heroes.forEach((h, i) => {
                            const wrColor = h.wr >= 60 ? 'var(--success)' : h.wr >= 45 ? 'var(--warning)' : 'var(--danger)';
                            html += `
                                <div class="draft-item">
                                    <div class="draft-medal">${medals[i] || ''}</div>
                                    <div class="draft-name">${h.hero}</div>
                                    <div class="draft-wr" style="color:${wrColor}">${h.wr}%</div>
                                    <div class="draft-picks">${h.picks} p</div>
                                </div>
                            `;
                        });
                    }
                    html += `</div>`;
                }
                document.getElementById('draftContent').innerHTML = html;
            }).getDraftHelperData();
        }

        // Games History
        function loadGames() {
            document.getElementById('gamesContent').innerHTML = '<div class="full-loader"><div class="spinner"></div></div>';
            google.script.run.withSuccessHandler(games => {
                if(!games.length) { document.getElementById('gamesContent').innerHTML = '<div class="full-loader">No matches yet</div>'; return; }
                
                let html = '';
                games.forEach(g => {
                    const isWin = g.result === 'Win';
                    html += `
                        <div class="game-card">
                            <div class="game-badge-res ${isWin ? 'tl-w' : 'tl-l'}"></div>
                            <div class="game-info">
                                <div class="game-top">
                                    <span class="g-tag ${isWin ? 'g-win' : 'g-lose'}">${g.result}</span>
                                    <span class="g-tag g-mode">${g.mode}</span>
                                    <span class="g-date">#${g.num} &bull; ${g.date}</span>
                                </div>
                                <div class="game-lineup">${g.lineup}</div>
                            </div>
                            <button class="btn-icon" onclick="deleteGame(${g.num})"><i class="fa-solid fa-trash-can"></i></button>
                        </div>
                    `;
                });
                document.getElementById('gamesContent').innerHTML = html;
            }).getGamesList();
        }

        // Actions
        function runBackend(fn) {
            showToast('Processing...');
            google.script.run.withSuccessHandler(() => showToast('Operation successful!'))
                .withFailureHandler(e => showToast(e.message, false))[fn]();
        }

        function deleteGame(num) {
            if(!confirm(`Delete Match #${num}?`)) return;
            showToast('Deleting...');
            google.script.run.withSuccessHandler(r => {
                showToast(r.message, r.success);
                if(r.success) loadGames();
            }).withFailureHandler(e => showToast(e.message, false)).deleteGameFromSidebar(num);
        }

        function uploadCsv() {
            const fi = document.getElementById('csvFile');
            if(!fi.files.length) { showToast('Select a CSV file first', false); return; }
            showToast('Uploading...');
            const reader = new FileReader();
            reader.onload = e => {
                google.script.run.withSuccessHandler(r => {
                    showToast(r.message, r.success);
                    if(r.success) { fi.value = ''; loadGames(); }
                }).withFailureHandler(err => showToast(err.message, false)).processCsvImport(e.target.result);
            };
            reader.readAsText(fi.files[0]);
        }

        // Modal Logic
        function openModal() {
            document.getElementById('modalAdd').classList.add('show');
            if(_players.length === 0) {
                google.script.run.withSuccessHandler(p => {
                    _players = p;
                    google.script.run.withSuccessHandler(h => {
                        _heroes = h;
                        buildFormLineup();
                    }).getHeroesList();
                }).getPlayersList();
            }
        }

        function closeModal() {
            document.getElementById('modalAdd').classList.remove('show');
        }

        function buildFormLineup() {
            let pOpts = '<option value="">Select Player...</option>' + _players.map(p => `<option value="${p}">${p}</option>`).join('');
            let html = '';
            for(let i=1; i<=5; i++) {
                html += `
                    <div class="p-row">
                        <select id="fp${i}">${pOpts}</select>
                        <input type="text" id="fh${i}" placeholder="Hero..." list="heroListDL">
                    </div>
                `;
            }
            html += `<datalist id="heroListDL">${_heroes.map(h => `<option value="${h}">`).join('')}</datalist>`;
            document.getElementById('formLineup').innerHTML = html;
        }

        function saveMatch() {
            const fd = {
                mode: document.getElementById('iMode').value,
                result: document.getElementById('iResult').value,
                duration: document.getElementById('iDur').value,
                notes: document.getElementById('iNotes').value
            };
            let hasData = false;
            for(let i=1; i<=5; i++) {
                const p = document.getElementById('fp'+i);
                const h = document.getElementById('fh'+i);
                if(p && h) {
                    fd['player'+i] = p.value;
                    fd['hero'+i] = h.value;
                    if(p.value && h.value) hasData = true;
                }
            }
            if(!hasData) { showToast('Select at least 1 player and hero!', false); return; }
            
            const btn = document.getElementById('btnSave');
            btn.disabled = true;
            btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Saving...';
            
            google.script.run.withSuccessHandler(r => {
                btn.disabled = false;
                btn.innerHTML = '<i class="fa-solid fa-check"></i> Save Match';
                showToast(r.message, r.success);
                if(r.success) {
                    for(let i=1; i<=5; i++) { document.getElementById('fp'+i).value=''; document.getElementById('fh'+i).value=''; }
                    document.getElementById('iNotes').value='';
                    closeModal();
                    if(document.getElementById('pg-games').classList.contains('active')) loadGames();
                }
            }).withFailureHandler(e => {
                btn.disabled = false;
                btn.innerHTML = '<i class="fa-solid fa-check"></i> Save Match';
                showToast(e.message, false);
            }).processManualAddGame(fd);
        }
